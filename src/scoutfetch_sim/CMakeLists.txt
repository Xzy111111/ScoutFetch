cmake_minimum_required(VERSION 3.8)
project(scoutfetch_sim)
find_package(ament_cmake REQUIRED)
install(DIRECTORY config launch rviz worlds models DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/gripper_adapter.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/scan_self_filter.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/sim_model.py DESTINATION share/${PROJECT_NAME}/scripts)
if(BUILD_TESTING)

  ament_add_pytest_test(config_contract test/test_config_contract.py)
  ament_add_pytest_test(gripper_adapter_contract test/test_gripper_adapter_contract.py)
  ament_add_pytest_test(sim_model test/test_sim_model.py)
  ament_add_pytest_test(scan_self_filter test/test_scan_self_filter.py)
endif()
ament_package()
